        if (p.life <= 0 || p.y < -60 || p.y > H + 60) {
          particles[i] = makeParticle(mode);
          continue;
        }

        const flicker = p.kind === "ember"
          ? 0.55 + Math.abs(Math.sin(p.phase * 3)) * 0.45
          : 0.7 + Math.abs(Math.sin(p.phase * 2)) * 0.3;

        const a = Math.max(0, Math.min(1, p.life)) * flicker;

        if (mode === "hell") {
          const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
          g.addColorStop(0, `hsla(${p.hue}, 100%, 62%, ${a})`);
          g.addColorStop(1, `hsla(${p.hue}, 100%, 50%, 0)`);
          ctx.fillStyle = g;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size * 4, 0, Math.PI * 2);
          ctx.fill();
        } else {
          ctx.fillStyle = `hsla(${p.hue}, 90%, ${Math.random() < 0.35 ? 100 : 68}%, ${a * 0.9})`;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
          ctx.fill();
        }
      }
      requestAnimationFrame(tick);
    }

    if (!reducedMotion) {
      tick();
    } else {
      ctx.globalCompositeOperation = "source-over";
      for (const p of particles) {
        ctx.fillStyle = `hsla(45, 80%, 70%, ${p.life * 0.6})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
      }
    }
  }

  /* ================= SCROLL PROGRESS ================= */
  const bar = $("#progressBar");
  if (bar) {
    function onScroll() {
      const max = document.documentElement.scrollHeight - window.innerHeight;
      bar.style.width = (max > 0 ? (window.scrollY / max) * 100 : 0) + "%";
    }
    window.addEventListener("scroll", onScroll, { passive: true });
    onScroll();
  }

  /* ================= HERO PARALLAX (home only) ================= */
  if (!reducedMotion && finePointer) {
    const heroContent = $(".hero-content");
    const orbs = $$(".orb");
    if (heroContent) {
      window.addEventListener("mousemove", (e) => {
        const dx = e.clientX / window.innerWidth - 0.5;
        const dy = e.clientY / window.innerHeight - 0.5;
        heroContent.style.transform = `translate(${dx * -14}px, ${dy * -10}px)`;
        orbs.forEach((o, i) => {
          o.style.translate = `${dx * (i ? 26 : -26)}px ${dy * (i ? 20 : -20)}px`;
        });
      }, { passive: true });
    }
  }
})();
